  <script>
    const SCRIPT_URL = 'https://script.google.com/macros/s/AKfycbzTqm8O4LOuXBv_8Hn3RP3uVdjmHlyZ9K5cf0GCMGzYA4YhzMgxP9X_RfpRpKotrt5pVA/exec';

    if ('serviceWorker' in navigator) {
      navigator.serviceWorker.register('sw.js');
    }

    // Auto-fill saved customer details on repeat visits
    window.onload = () => {
      if (localStorage.getItem('wassers_name')) document.getElementById('name').value = localStorage.getItem('wassers_name');
      if (localStorage.getItem('wassers_phone')) {
        document.getElementById('phone').value = localStorage.getItem('wassers_phone');
        document.getElementById('lookupPhone').value = localStorage.getItem('wassers_phone');
      }
      if (localStorage.getItem('wassers_area')) document.getElementById('area').value = localStorage.getItem('wassers_area');
    };

    function switchTab(tab) {
      document.getElementById('tabOrder').classList.toggle('active', tab === 'order');
      document.getElementById('tabHistory').classList.toggle('active', tab === 'history');
      document.getElementById('viewOrder').classList.toggle('hidden', tab !== 'order');
      document.getElementById('viewHistory').classList.toggle('hidden', tab !== 'history');
      if (tab === 'history' && document.getElementById('lookupPhone').value.length === 10) fetchHistory();
    }

    function calcTotal() {
      const sel = document.getElementById('product');
      const price = Number(sel.options[sel.selectedIndex].dataset.price);
      const qty = Number(document.getElementById('qty').value) || 1;
      document.getElementById('totalDisplay').innerText = '₹' + (price * qty);
      return price * qty;
    }

    function placeOrder() {
      const name = document.getElementById('name').value.trim();
      const phone = document.getElementById('phone').value.replace(/\D/g, '').slice(-10);
      const area = document.getElementById('area').value;
      const product = document.getElementById('product').value;
      const qty = document.getElementById('qty').value;
      const total = calcTotal();

      if (!name || phone.length !== 10) {
        alert('Please enter your Business/Customer Name and a valid 10-digit Mobile Number.');
        return;
      }

      // Save customer details on their phone for 1-tap future reorders
      localStorage.setItem('wassers_name', name);
      localStorage.setItem('wassers_phone', phone);
      localStorage.setItem('wassers_area', area);
      document.getElementById('lookupPhone').value = phone;

      // Save to Google Sheets
      fetch(SCRIPT_URL, {
        method: 'POST',
        mode: 'no-cors',
        headers: { 'Content-Type': 'text/plain' },
        body: JSON.stringify({ name, phone, area, product, qty, total })
      });

      // Open WhatsApp to 8469277377
      const msg = `*NEW WASSERS & CO. ORDER*%0A` +
                  `• *Client:* ${name}%0A` +
                  `• *Phone:* ${phone}%0A` +
                  `• *Zone:* ${area}%0A` +
                  `• *Item:* ${product}%0A` +
                  `• *Qty:* ${qty}%0A` +
                  `• *Total:* ₹${total}`;

      window.open(`https://wa.me/918469277377?text=${msg}`, '_blank');
    }

    async function fetchHistory() {
      const phone = document.getElementById('lookupPhone').value.replace(/\D/g, '').slice(-10);
      const box = document.getElementById('historyResults');
      if (phone.length !== 10) {
        alert('Enter a valid 10-digit mobile number.');
        return;
      }
      box.innerHTML = '<p style="color:#94A3B8;font-size:13px;">Loading your orders...</p>';
      try {
        const res = await fetch(`${SCRIPT_URL}?phone=${phone}`);
        const orders = await res.json();
        if (!orders.length) {
          box.innerHTML = '<p style="color:#94A3B8;font-size:13px;">No orders found for this number yet.</p>';
          return;
        }
        box.innerHTML = orders.map(o => `
          <div class="order-item">
            <span class="badge">${o.status}</span>
            <strong style="color:#C5A059;">${o.product} × ${o.qty}</strong><br/>
            <span style="color:#CBD5E1;">Total: ₹${o.total} • ${o.area}</span><br/>
            <span style="color:#64748B;font-size:11px;">${o.date}</span>
          </div>
        `).join('');
      } catch (err) {
        box.innerHTML = '<p style="color:#F87171;font-size:13px;">Could not load history. Please try again.</p>';
      }
    }
  </script>
